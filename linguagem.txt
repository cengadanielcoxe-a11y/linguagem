
numero = 0
nome = ""
idade = ""
sexo = ""
endereco = ""
nif = ""

data = ""
diagnostico = ""
observacoes = ""

while True:

    print("\n===== SISTEMA DE UTENTES =====")
    print("1 - Registar utente")
    print("2 - Registar consulta")
    print("3 - Consultar historico")
    print("4 - Sair")

    opcao = input("Escolha uma opcao: ")

    if opcao == "1":

        if numero == 0:

            print("\n--- REGISTO DE UTENTE ---")

            nome = input("Nome: ")
            idade = input("Idade: ")
            sexo = input("Sexo: ")
            endereco = input("Endereco: ")
            nif = input("NIF: ")

            numero = 1

            print("\nUtente registado com sucesso!")
            print("Numero de registo:", numero)

        else:
            print("\nJa existe um utente registado!")

    elif opcao == "2":

        if numero != 0:

            print("\n--- REGISTAR CONSULTA ---")

            data = input("Data da consulta: ")
            diagnostico = input("Diagnostico: ")
            observacoes = input("Observacoes: ")

            print("\nConsulta registada com sucesso!")

        else:
            print("\nPrimeiro deve registar um utente!")

    elif opcao == "3":

        print("\n--- CONSULTAR UTENTE ---")

        pesquisa = int(input("Introduza o numero de registo: "))

        if pesquisa == numero and numero != 0:

            print("\n--- DADOS DO UTENTE ---")

            print("Numero:", numero)
            print("Nome:", nome)
            print("Idade:", idade)
            print("Sexo:", sexo)
            print("Endereco:", endereco)
            print("NIF:", nif)

            print("\n--- HISTORICO CLINICO ---")

            if data == "":
                print("Nao existem consultas registadas.")

            else:
                print("Data:", data)
                print("Diagnostico:", diagnostico)
                print("Observacoes:", observacoes)

        else:
            print("\nUtente nao encontrado!")

    elif opcao == "4":

        print("\nPrograma encerrado!")
        break

    else:
        print("\nOpcao invalida!")
